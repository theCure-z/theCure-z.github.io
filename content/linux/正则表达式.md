# 普通字符

abc->直接匹配"abc"

# 点号.

a.c->匹配"abc" "a1c" "a-c"

# 字符类 []

[abc] 匹配任意一个 "a"/"b"/"c"
[a-z] 所有小写字母
[0-9] 所有数字

# 取反 [^ ]

[^0-9] 所有非数字

# 重复

*0 次或多次
+1 次或多次
？0 次或 1 次

+ ab* -> 匹配 a ab abb abbbb
+ ab+ -> 匹配 ab abbb

# 分组()

(ab)+ -> 匹配 ab abab

# 或 |

cat|dog -> 匹配 cat 或 dog

# 边界

^ 行开头
$ 行结尾
\b 单词边界

+ ^abc -> 匹配行首为 abc
+ abc$ -> 匹配行尾为 abc
